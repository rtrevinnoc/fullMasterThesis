\section{Antecedentes}

\begin{frame}{Antecedentes: revisión de literatura}
    \begin{purpleblock}{Estrategia de revisión}
        Análisis de literatura organizado en tres ejes: (i) \emph{control de movimiento} de etapas litográficas, (ii) \emph{planificación de trayectorias} de orden superior y (iii) \emph{modelado de rendimiento} basado en mapas de defectos.
    \end{purpleblock}
    \vspace{0.2cm}
    \begin{columns}
        \column{.5\textwidth}
        \textbf{Control y dinámica}
        \begin{itemize}
            \item Modelos multicuerpo flexibles con frecuencias dependientes de la posición (\cite{Al-Rawashdeh2022_Caracterization}).
            \item Integradores no lineales (HIGS, CgLp) para mitigar el retraso de fase (\cite{Heertjes2025_NonlinearControl}).
        \end{itemize}
        \column{.5\textwidth}
        \textbf{Trayectorias}
        \begin{itemize}
            \item \textit{Input shaping} y trayectorias acotadas en \textit{snap} (\cite{Lambrechts2003_Trajectories,Al-Rawashdeh2023_Trajectories,Dong2023_SCurve}).
        \end{itemize}
        \textbf{Rendimiento}
        \begin{itemize}
            \item Modelos de \textit{yield} con aprendizaje sobre mapas de defectos (\cite{Kim2019_Productivity,Yang2018_Yield,Wu2015_WM811K}).
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Estado del arte en control de movimiento}
    La literatura ha consolidado modelos en los que las frecuencias naturales dependen de la posición instantánea del scanner.
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Control activo:} integradores no lineales (HIGS, CgLp) para reducir el sobretiro (\cite{Heertjes2025_NonlinearControl}).
        \item \textbf{Arquitecturas dinámicas:} configuración de triple etapa (\textit{fine stages}) (\cite{Al-Rawashdeh2022_Caracterization}).
        \item \textbf{\textit{Input shaping}:} trayectorias de referencia acotadas en el \textit{jerk} o \textit{snap} para evitar la excitación de modos resonantes (\cite{Al-Rawashdeh2023_Trajectories,Lambrechts2003_Trajectories}).
    \end{itemize}
    \vspace{0.2cm}
    Estos enfoques optimizan \textbf{métricas servo-mecánicas} (MSE, tiempo de asentamiento) y se mantienen \emph{aislados} del proceso que pretenden controlar: la impresión del circuito integrado.
\end{frame}

\section{Identificación del vacío}

\begin{frame}{La desconexión entre trayectoria y \textit{yield}}
    \begin{redblock}{Vacío identificado}
        La industria carece de una metodología que, de forma \textbf{predictiva}, traduzca los límites de la trayectoria $(v_{\max}, a_{\max}, j_{\max}, s_{\max})$ en la morfología del error sobre el patrón impreso, cerrando el lazo entre la planificación cinemática y el rendimiento.
    \end{redblock}
    \vspace{0.15cm}
    \begin{itemize}
        \item La literatura de control optimiza métricas servo-mecánicas (MSE, \textit{settling time}) que \emph{no} discriminan entre patrones de vibración con \textbf{forma distinta} sobre la oblea.
        \item Los modelos de \textit{yield} basados en aprendizaje (\cite{Yang2018_Yield,Wu2015_WM811K}) clasifican defectos \emph{después} del proceso; nunca se realimentan a la planificación de la trayectoria.
        \item Las optimizaciones de mapa de oblea (\cite{Kim2017_Market,Kim2019_Productivity}) maximizan \textit{throughput} suponiendo perfiles cinemáticos fijos y no consideran cómo cambian los defectos al modificarlos.
    \end{itemize}
\end{frame}

\begin{frame}{Consecuencias del vacío}
    \begin{columns}
        \column{.55\textwidth}
        \begin{itemize}
            \item Múltiples patrones de vibración producen \textbf{formas distintas} de error sobre la oblea; minimizar el MSE no captura esa diferencia.
            \item Sin un modelo predictivo embebido en la planeación, la correlación entre el perfil de movimiento y el impacto sobre la oblea \textbf{permanece desconocida} hasta después de la fabricación.
            \item Esto obliga a usar parámetros conservadores y limita el avance hacia una \emph{manufactura inteligente} consciente del contexto del proceso.
            \item \textbf{Falta un puente predictivo} que cierre el lazo entre la planificación cinemática y el rendimiento de la oblea.
        \end{itemize}
        \column{.45\textwidth}
        \centering

\section{Planteamiento del problema}

\begin{frame}{Definición del problema}
    \begin{redblock}{Problema}
        En un scanner litográfico, los parámetros cinemáticos de la trayectoria del \textit{wafer/reticle stage} $(v_{\max}, a_{\max}, j_{\max}, s_{\max})$ se eligen para maximizar \textit{throughput}, pero su impacto sobre la \textbf{morfología espacial de defectos} en la oblea no se conoce \emph{a priori}. Esta ausencia de un modelo predictivo obliga a usar parámetros conservadores y produce un compromiso \textit{throughput--yield} subóptimo.
    \end{redblock}
    \vspace{0.15cm}
    \textbf{Elementos del problema}
    \begin{itemize}
        \item \textbf{Sujeto:} sistema multicuerpo flexible del \textit{stage} (cadenas retícula--óptica--oblea).
        \item \textbf{Variable de decisión:} perfil de aceleración de cuarto orden.
        \item \textbf{Restricción:} la \textit{productividad} no debe degradarse respecto a la operación nominal.
        \item \textbf{Criterio:} maximización del rendimiento (\textit{yield}) evaluado sobre el mapa de defectos espacial.
    \end{itemize}
\end{frame}

\section{Preguntas de investigación}

\begin{frame}{Preguntas de investigación}
    \begin{purpleblock}{Pregunta general}
        ¿Cómo deben elegirse los parámetros cinemáticos de una trayectoria de cuarto orden para que el rendimiento (\textit{yield}) de la oblea se maximice sin penalizar la productividad (\textit{throughput})?
    \end{purpleblock}
    \vspace{0.15cm}
    \textbf{Preguntas específicas}
    \begin{enumerate}
        \item ¿Cómo se relacionan los parámetros $(v_{\max}, a_{\max}, j_{\max}, s_{\max})$ con el \textbf{error de seguimiento dinámico} y el \textbf{error de sincronización} de las cadenas retícula--oblea?
        \item ¿Es posible \emph{predecir} la morfología del mapa de defectos a partir del error de seguimiento generado por un perfil cinemático dado?
        \item ¿Qué arquitectura y configuración de \textbf{NN} permite estimar el rendimiento (\textit{yield}) con suficiente exactitud para servir como función de costo en una optimización?
    \end{enumerate}
\end{frame}

\section{Hipótesis}

\begin{frame}{Hipótesis}
    \begin{blueblock}{Hipótesis}
        Es posible maximizar conjuntamente el \textit{yield} y la productividad de máquinas de litografía si los perfiles de trayectoria de \textbf{cuarto orden} se optimizan con un algoritmo heuristico que utiliza una \textbf{NN} entrenada sobre mapas de defectos como función de costo predictiva, evaluada sobre los mapas que produce una \textit{simulación} del scanner.
    \end{blueblock}
    \vspace{0.2cm}
    \textbf{Supuestos verificables}
    \begin{itemize}
        \item El error de seguimiento dinámico simulado es representativo del error de impresión real sobre la oblea.
        \item La CNN entrenada sobre WM-811K generaliza a mapas sintéticos derivados del modelo.
        \item El espacio de parámetros cinemáticos es explorable por un algoritmo heuristico en tiempos razonables.
    \end{itemize}
\end{frame}

\section{Objetivos}

\begin{frame}{Objetivo general}
    \begin{purpleblock}{Objetivo general}
        Desarrollar una metodología \textbf{predictiva} de planificación de trayectorias de cuarto orden para scanners litográficos que, mediante la integración de un modelo subrogado de defectos basado en NN, permita \textbf{encontrar perfiles de aceleración óptimos} que maximicen el \textit{yield} sin penalizar el \textit{throughput}.
    \end{purpleblock}
\end{frame}

\begin{frame}{Objetivos específicos}
    \begin{enumerate}
        \item \textbf{Mejorar el seguimiento dinámico} de la trayectoria de referencia en el sistema multicuerpo, reduciendo el error de seguimiento y el error de sincronización entre las cadenas retícula--oblea bajo perfiles de cuarto orden.\\[-0.05cm]
        \item \textbf{Establecer el vínculo predictivo} entre los parámetros cinemáticos $(v_{\max}, a_{\max}, j_{\max}, s_{\max})$ y la morfología del mapa de defectos, formulando una función de costo que combine \textit{throughput} y \textit{yield} estimado por la NN sobre los mapas generados por la simulación.\\[-0.05cm]
        \item \textbf{Encontrar perfiles de aceleración óptimos} de cuarto orden que maximicen el \textit{yield} sin penalizar la productividad, mediante optimización con un algoritmo bio-inspirado heuristico sobre la función de costo.
    \end{enumerate}
    \vspace{0.15cm}
    \scriptsize El \emph{modelo dinámico multicuerpo} y la \emph{CNN subrogada} no son objetivos sino \textbf{herramientas} de la metodología; se describen en la sección correspondiente.
\end{frame}

\section{Variables}

\begin{frame}{Variables del estudio}
    \begin{columns}[T]
        \column{.5\textwidth}
        \textbf{Variables independientes}
        \begin{itemize}
            \item Tiempos de los segmentos de la trayectoria.
            \item Límites cinemáticos $v_{\max},\; a_{\max},\; j_{\max},\; s_{\max}$.
            \item \textbf{Arquitectura de la NN} (capas convolucionales, filtros, tamaño de entrada).
        \end{itemize}
        \vspace{0.15cm}
        \textbf{Variables intervinientes}
        \begin{itemize}
            \item Parámetros del modelo multicuerpo ($K,\,C,\,m$ por cuerpo).
            \item Patrón de exposición de la oblea.
        \end{itemize}
        \column{.5\textwidth}
        \textbf{Variables dependientes}
        \begin{itemize}
            \item Error de seguimiento dinámico.
            \item \textbf{Error de sincronización} retícula--oblea.
            \item \textbf{Mapa de defectos} generado por la simulación.
            \item Rendimiento de manufactura (\textit{yield}) estimado por la NN.
            \item Productividad (\textit{throughput}).
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Retos identificados}
    \begin{columns}[T]
        \column{.5\textwidth}
        \textbf{Retos técnicos}
        \begin{itemize}
            \item Tiempo de convergencia del PSO sobre la función de costo (simulación + NN por evaluación).
            \item Generalización de la NN (entrenada en WM-811K real) a mapas \emph{sintéticos} producidos por simulación.
        \end{itemize}
        \column{.5\textwidth}
        \textbf{Retos metodológicos}
        \begin{itemize}
            \item Definir el umbral de aceptación de $\widehat{\text{yield}}$ que sirve como referencia.
            \item Trazabilidad entre los parámetros cinemáticos y la forma del defecto.
        \end{itemize}
    \end{columns}
    \vspace{0.2cm}
    \begin{redblock}{Riesgo principal}
        La complejidad de la función de costo dinámica (modelo multicuerpo acoplado a la NN) puede degradar los tiempos de convergencia del algoritmo heuristico. \emph{Mitigación:} sustitución parcial por modelo reducido o uso de evaluaciones segmentadas.
    \end{redblock}
\end{frame}
